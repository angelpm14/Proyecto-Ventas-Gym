CREATE DATABASE GYMMEGAMUSCULO_VENTAS;

USE GYMMEGAMUSCULO_VENTAS;

CREATE TABLE Productos(
    id INT AUTO_INCREMENT,
    nombre VARCHAR(100),
    descripcion TEXT,
    precio DECIMAL(10,2),
    PRIMARY KEY(id)
);

CREATE TABLE Clientes(
    id INT AUTO_INCREMENT,
    nombre VARCHAR(100),
    apellido VARCHAR(100),
    correo_electronico VARCHAR(100),
    telefono VARCHAR(15),
    PRIMARY KEY(id)
);

CREATE TABLE Ventas(
    id_Ventas INT Identity,
    id_cliente int,
    id_Usuario int,
    tipo_comprob varchar(20),
    serie_comprob varchar(7),
    num_comprob varchar(10),
    fecha_venta DATEtime,
    total decimal(9,2)
    PRIMARY KEY(id_Ventas),
    FOREIGN KEY(id_cliente) REFERENCES Cliente(id_cliente),
    FOREIGN KEY(id_Usuario) REFERENCES Usuario(id_Usuario)
);


CREATE TABLE Detall_Ventas(
    id_detalVent INT AUTO_INCREMENT,
    id_Ventas int,
    cantidad INT,
    precio decimal(9,2),
    descuent decmal(9,2)
    PRIMARY KEY(id_Ventas),
    FOREIGN KEY(cliente_id) 
    REFERENCES Clientes(id)
);

{
    bool Agregar = true;
    foreach (DataRow FilaTemp in DDetalle.Rows)
    {
        if (Convert.ToInt32(FilaTemp["ID_PROD"]) == ID_PROD)
        {
            Agregar = false;
            MensajeError("El artículo ya ha sido agregado.");
        }
    }
    if (Agregar)
    {
        DataRow Fila = DDetalle.NewRow();
        Fila["ID_PROD"] = ID_PROD;
        Fila["CODIGO"] = COD_PROD;
        Fila["NOMBRE"] = NOMB_PROD;
        Fila["STOCK"] = STOCK_PROD;
        Fila["CANTIDAD"] = 1;
        Fila["PRECIO"] = PREC_DETALLVENT;
        Fila["DESCUENTO"] = 0;
        Fila["IMPORTE"] = PREC_DETALLVENT;
        DDetalle.Rows.Add(Fila);
        CalcularTotales();
    }
